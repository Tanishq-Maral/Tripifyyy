# Import the necessary libraries
from qiskit import QuantumCircuit,QuantumRegister,ClassicalRegister
from qiskit_aer import AerSimulator
import matplotlib.pyplot as plt

# Create the error-correcting circuit
def create_error_correction_circuit(logical_bit,error_qubit):
    # Three data quibits and two syndrome ancillas
    data = QuantumRegister(3, 'data')
    ancilla = QuantumRegister(2, 'ancilla')

    syndrome = ClassicalRegister(2, 'syndrome')
    result = ClassicalRegister(3, 'result')

    qc = QuantumCircuit(data, ancilla, syndrome, result)

    # Step1:Encode the logical bit
    if logical_bit==1:
        qc.x(data[0])
        qc.x(data[1])
        qc.x(data[2])

    qc.barrier()

    # Step 2: Introduce an bit flip error
    qc.x(data[error_qubit])

    qc.barrier()

    # Step 3:Measure parity to identify the error
    qc.cx(data[0], ancilla[0])
    qc.cx(data[1], ancilla[0])

    qc.cx(data[1], ancilla[1])
    qc.cx(data[2], ancilla[1])

    qc.measure(ancilla[0], syndrome[0])
    qc.measure(ancilla[1], syndrome[1])
    qc.barrier()

    # Step4:Correct the error using the syndrome
    # Syndrome 01(integer1):qubit 0 flipped
    with qc.if_test((syndrome, 1)):
        qc.x(data[0])

    # Syndrome 11(integer3):qubit 1 flipped
    with qc.if_test((syndrome, 3)):
        qc.x(data[1])

    # Syndrome 10(integer2):qubit 2 flipped
    with qc.if_test((syndrome, 2)):
        qc.x(data[2])

    qc.barrier()

    # Step5:Measure the data qubits to get the result
    qc.measure(data, result)

    return qc

Tarrataca’s Quantum Production System for the 3-Puzzle Problem

# Import the necessary libraries
import numpy as np
import matplotlib.pyplot as plt

from qiskit import QuantumCircuit, QuantumRegister, ClassicalRegister
from qiskit.circuit.library import UnitaryGate
from qiskit_aer import AerSimulator

# Define the 3 puzzle problem as a unitary matrix
# 0->blank
# 1,2,3->numbered tiles

initial_state = (1,0,3,2)
goal_state = (1,2,3,0)

def display_board(state):
    print(f"{state[0]} {state[1]}")
    print(f"{state[2]} {state[3]}")
    print()

print("Initial State:")
display_board(initial_state)

print("Goal State:")
display_board(goal_state)

def apply_rule(state, move):
    state = list(state)

    # Positions around the board in clockwise order
    cycle = [0, 1, 3, 2]

    blank_position = state.index(0)
    cycle_position = cycle.index(blank_position)

    if move == 1:
        # Clockwise
        next_position = cycle[(cycle_position + 1) % 4]
    else:
        # Counterclockwise
        next_position = cycle[(cycle_position - 1) % 4]

    state[blank_position], state[next_position] = (
        state[next_position],
        state[blank_position]
    )

    return tuple(state)


print("Counterclockwise result:")
display_board(apply_rule(initial_state, 0))

print("Clockwise result:")
display_board(apply_rule(initial_state, 1))

# Encode the board into binary
def encode_board(state):
    value = 0
    for position, tile in enumerate(state):
        value |= (tile<<(2*position))

    return value


def decode_board(value):
    return tuple(
        (value>>(2*position))&3
        for position in range(4)
    )


initial_code = encode_board(initial_state)
goal_code = encode_board(goal_state)

print("initial binary encoding:",format(initial_code,'08b'))
print("goal binary encoding:",format(goal_code,'08b'))
print("Decoded initial state:",decode_board(initial_code))

# Construct the reversible production operator

BOARD_DIM = 256
MOVE_DIM = 4
TOTAL_DIM = BOARD_DIM * MOVE_DIM


def next_board_code(board_code, move):
    state = decode_board(board_code)

    # Check whether the board contains all three tiles and the blank
    if sorted(state) != [0, 1, 2, 3]:
        return board_code

    return encode_board(apply_rule(state, move))


move_matrix = np.zeros((TOTAL_DIM, TOTAL_DIM), dtype=complex)

for move_code in range(MOVE_DIM):
    for board_code in range(BOARD_DIM):

        if move_code in (0, 1):
            new_board = next_board_code(board_code, move_code)
        else:
            new_board = board_code

        input_index = board_code + move_code * BOARD_DIM
        output_index = new_board + move_code * BOARD_DIM

        move_matrix[output_index, input_index] = 1


production_gate = UnitaryGate(
    move_matrix,
    label="Reversible Rules"
)

print("Production gate qubits:", production_gate.num_qubits)
print("Expected: 10")

# Create the target state oracle
oracle_matrix = np.eye(TOTAL_DIM, dtype=complex)

for move_code in range(MOVE_DIM):
    input_index = goal_code + move_code * BOARD_DIM
    oracle_matrix[input_index, input_index] = -1

oracle_gate = UnitaryGate(
    oracle_matrix,
    label="Goal Oracle"
)

print("Goal oracle created.")

# Build the quantum production system 
def build_circuit(use_grover=True):
    board = QuantumRegister(8, name="board")
    move = QuantumRegister(2, name="move")
    output = ClassicalRegister(10, name="output")

    qc = QuantumCircuit(board, move, output)

    for position,tile in enumerate(initial_state):
        for bit in range(2):
            if (tile>>bit)&1:
                qc.x(board[2*position+bit])


    qc.h(move[0])
    qc.h(move[1])

    all_qubits = list(board) + list(move)

    qc.append(production_gate, all_qubits)

    if use_grover:
        qc.append(oracle_gate, all_qubits)
        qc.append(production_gate.inverse(),all_qubits)

        qc.h(move)
        qc.x(move)
        qc.cz(move[0], move[1])
        qc.x(move)
        qc.h(move)

        qc.append(production_gate, all_qubits)

    qc.measure(board, output[0:8])
    qc.measure(move, output[8:10])

    return qc


qc = build_circuit(use_grover=True)
print(qc.draw(output="text",fold=120))

# Execute the circuit
simulator = AerSimulator()

job = simulator.run(qc, shots=1024)
counts = job.result().get_counts()

print("Measurement Results:")
print(counts)

goal_shots = 0
move_counts = [0, 0, 0, 0]

for bitstring, count in counts.items():
    board_code = int(bitstring[-8:], 2)
    move_code = int(bitstring[:2], 2)

    move_counts[move_code] += count

    print(
        f"Move choice: {move_code:02b}, "
        f"Board: {decode_board(board_code)}, "
        f"Count: {count}"
    )

    if board_code == goal_code:
        goal_shots += count

success_rate = goal_shots / sum(counts.values()) * 100

print("\nGoal state:", goal_state)
print("Successful shots:", goal_shots)
print(f"Goal-state success rate: {success_rate:.2f}%")

baseline_circuit = build_circuit(use_grover=False)

baseline_counts = simulator.run(
    baseline_circuit,
    shots=1024
).result().get_counts()


def calculate_goal_rate(counts):
    successful = sum(
        count
        for bitstring, count in counts.items()
        if int(bitstring[-8:], 2) == goal_code
    )

    total = sum(counts.values())
    return 100 * successful / total


baseline_rate = calculate_goal_rate(baseline_counts)

print(f"Without Grover amplification: {baseline_rate:.2f}%")
print(f"With Grover amplification:    {success_rate:.2f}%")

plt.bar(
    ["Without Grover", "With Grover"],
    [baseline_rate, success_rate]
)

plt.ylabel("Goal-state probability (%)")
plt.title("Quantum Production System: Search Comparison")
plt.ylim(0, 105)
plt.show()

Implement Quantum Teleportation algorithm in Python

import numpy as np
import matplotlib.pyplot as plt

from qiskit import QuantumCircuit, transpile
from qiskit_aer import AerSimulator
from qiskit.visualization import plot_histogram

# Angles used to prepare the input state
theta = np.pi / 3
phi = np.pi / 4

# Three qubits and three classical bits:
# c[0], c[1] store Alice's measurements; c[2] stores verification
qc = QuantumCircuit(3, 3)

# Step 1: Prepare the state to teleport on qubit 0
qc.ry(theta, 0)
qc.rz(phi, 0)
qc.barrier()

# Step 2: Create an entangled Bell pair between Alice (qubit 1) and Bob (qubit 2)
qc.h(1)
qc.cx(1, 2)
qc.barrier()

# Step 3: Alice performs the Bell-measurement operations
qc.cx(0, 1)
qc.h(0)
qc.barrier()

# Step 4: Alice measures qubits 0 and 1
qc.measure(0, 0)
qc.measure(1, 1)
qc.barrier()

# Step 5: Bob corrects his qubit using Alice's classical measurement results
with qc.if_test((qc.clbits[1], 1)):
    qc.x(2)

with qc.if_test((qc.clbits[0], 1)):
    qc.z(2)

qc.barrier()

# Step 6: Verify the teleported state.
# Undo the original preparation on Bob's qubit.
qc.rz(-phi, 2)
qc.ry(-theta, 2)
qc.measure(2, 2)

qc.draw(output="mpl")

simulator = AerSimulator()
compiled_qc = transpile(qc, simulator)

result = simulator.run(compiled_qc, shots=1000).result()
counts = result.get_counts()

print("Measurement results:", counts)

plot_histogram(counts)
plt.show()

total_shots = sum(counts.values())
verification_ones = sum(count for bitstring, count in counts.items()
                         if bitstring.replace(" ", "")[0] == "1")
verification_zeros = total_shots - verification_ones

print("Total shots:", total_shots)
print("Verification results equal to 0:", verification_zeros)
print("Verification results equal to 1:", verification_ones)

if verification_ones == 0:
    print("Conclusion: Quantum teleportation was successful in this ideal simulation.")
else:
    print("Some verification results were 1. Check the circuit or simulator settings.")

## Conclusion
The experiment demonstrates quantum teleportation using entanglement, measurement, 
classical communication, and correction gates. The state is transferred to Bob's qubit; 
the original state is not copied, and classical communication is still required.

Implementing a 5 qubit Quantum Fourier Transform

import numpy as np
import matplotlib.pyplot as plt

from qiskit import QuantumCircuit, transpile
from qiskit.quantum_info import Statevector
from qiskit_aer import AerSimulator
from qiskit.visualization import plot_histogram

n = 5
qc = QuantumCircuit(n)

# Prepare an example input |00101> (decimal 5)
# Qiskit uses qubit 0 as the least-significant bit.
input_value = 5
for qubit in range(n):
    if (input_value >> qubit) & 1:
        qc.x(qubit)

qc.barrier()

# Apply the Quantum Fourier Transform
for target in range(n):
    qc.h(target)
    for control in range(target + 1, n):
        angle = np.pi / (2 ** (control - target))
        qc.cp(angle, control, target)

# Reverse qubit order to match the standard QFT output ordering
for qubit in range(n // 2):
    qc.swap(qubit, n - qubit - 1)

qc.draw(output="mpl")

state = Statevector.from_instruction(qc)
probabilities = state.probabilities()

labels = [format(i, "05b") for i in range(2**n)]

plt.figure(figsize=(12, 4))
plt.bar(labels, probabilities)
plt.xticks(rotation=90)
plt.xlabel("Computational basis state")
plt.ylabel("Probability")
plt.title("5-Qubit QFT Output Probabilities")
plt.tight_layout()
plt.show()

print("Number of qubits:", n)
print("Number of basis states:", 2**n)
print("Sum of probabilities:", round(float(np.sum(probabilities)), 6))

# Qiskit provides the standard QFT circuit through its library.
# Build an inverse-QFT circuit and apply it to the output.
inverse_qft = qc.copy()
# Remove the input-state preparation and keep only the QFT operations
# Rebuild the QFT-only circuit for a clear verification.
qft = QuantumCircuit(n)

for target in range(n):
    qft.h(target)
    for control in range(target + 1, n):
        angle = np.pi / (2 ** (control - target))
        qft.cp(angle, control, target)

for qubit in range(n // 2):
    qft.swap(qubit, n - qubit - 1)

test_circuit = QuantumCircuit(n)
for qubit in range(n):
    if (input_value >> qubit) & 1:
        test_circuit.x(qubit)

test_circuit.compose(qft, inplace=True)
test_circuit.compose(qft.inverse(), inplace=True)

recovered_state = Statevector.from_instruction(test_circuit)
recovered_probabilities = recovered_state.probabilities()

print("Probability of recovering the original state |00101>:", 
      round(float(recovered_probabilities[input_value]), 6))

## Conclusion
The experiment demonstrates the QFT using Hadamard gates, controlled 
phase-rotation gates, and SWAP gates. The inverse-QFT check confirms 
that the original input state can be recovered.

#Define the documents
documents = {
    "D1": "Python is easy to learn",
    "D2": "Python is used for machine learning",
    "D3": "Information retrieval uses an inverted index",
    "D4": "Python is useful for information retrieval"
}

print("Documents:")
for doc_id, text in documents.items():
    print(doc_id, ":", text)

# Create an inverted index
import re
inverted_index = {}
for doc_id, text in documents.items():
    # Tokenize the text into words
    words = re.findall(r'\b\w+\b', text.lower()) 

    for word in words:
        if word not in inverted_index:
            inverted_index[word]=[]
        if doc_id not in inverted_index[word]:
            inverted_index[word].append(doc_id)

print("\nInverted Index:\n")
for word, doc_ids in sorted(inverted_index.items()):
    print(word, ":", doc_ids)

# Search for a document

query = input("Enter a search word: ").lower().strip()

if query in inverted_index:
    print("Documents containing",query,":")
    for doc_id in inverted_index[query]:
        print(doc_id, ":", documents[doc_id])
else:
    print("No documents found containing")

# Support multiple words
query = input("Enter search words: ").lower().split()

if query:
    result_sets = [
        set(inverted_index.get(word, []))
        for word in query
    ]

    matching_docs = set.intersection(*result_sets)

    if matching_docs:
        print("Matching Documents:")
        for doc_id in sorted(matching_docs):
            print(doc_id, ":", documents[doc_id])
    else:
        print("No matching documents found.")

        return None
    if ":" in parsed_url.path[len("/wiki/"):]:  # Exclude special pages
        return None
    
    return "https://en.wikipedia.org" + parsed_url.path
    


queue = [normalise_url(start_url)]
visited = set()
page_titles = {}
extracted_links = {}

while queue and len(visited) < MAX_PAGES:
    url = queue.pop(0)
    if url in visited:
        continue

    try:
        response = session.get(url,timeout=15)
        response.raise_for_status()
        soup = BeautifulSoup(response.text, 'html.parser')

        title_tag=soup.find("title")
        title = title_tag.get_text(strip=True) if title_tag else url

        links = set()

        for tag in soup.find_all('a',href=True):
            full_url = urljoin(url, tag['href'])
            clean_url = normalise_url(full_url)

            if clean_url:
                links.add(clean_url)

        visited.add(url)
        page_titles[url] = title
        extracted_links[url] = links
        for link in sorted(links):
            if link not in visited and link not in queue:
                queue.append(link)
        print("Collected:",title)
    except Exception as e:
        print("Could not fetch:",url,"Error:",e)

print("Total pages collected:",len(visited))
# Build the direceted graph
pages = list(visited)
web_graph = {
    page :[
        link for link in extracted_links[page]
        if link in visited
    ]
    for page in pages
}

print("Webgraph:\n")
for page, links in web_graph.items():
    print(page_titles[page])
    print("Links to:",[page_titles[link] for link in links])
    print()

# Initalize PageRank values
N = len(pages)
damping_factor = 0.85
max_iterations = 100
tolerance = 1.e-8

if N==0:
    raise ValueError("No pages collected. Cannot compute PageRank.")


pagerank = {page: 1.0 / N for page in pages}
print("Number of pages:",N)
print("Damping factor:",damping_factor)
print("Initial PageRank values:",pagerank)

# Implement the PageRank algorithm
for iteration in range(max_iterations):
    new_pagerank ={
        page:(1-damping_factor)/N
        for page in pages
    }
    # Handling page with no outgoing links
    dangling_sum = sum(
        pagerank[page]
        for page in pages
        if len(web_graph[page])==0
    )

    for page in pages:
        new_pagerank[page] +=(
            damping_factor * dangling_sum / N
        )

    for source in pages:
        outgoing = web_graph[source]

        if outgoing:
            contribution = (
                damping_factor * pagerank[source] / len(outgoing)
            )
            for target in outgoing:
                new_pagerank[target] += contribution
    diff = sum(abs(new_pagerank[page]-pagerank[page]) for page in pages)
    pagerank = new_pagerank
    if diff < tolerance:
        break

print("Iterations completed:",iteration+1)
print("Page rank calculated successfully.")

# Display the final ranking 
ranked_pages = sorted(
    pagerank.items(),
    key=lambda x: x[1],
    reverse=True
)

print("Final PageRank Ranking:\n")
for rank, (page, score) in enumerate(ranked_pages, start=1):
    print(f"{rank}. {page_titles[page]} - PageRank: {score:.6f}")

# Visualize the Pagerank scores
top_pages = ranked_pages[:10]

labels = [
    page_titles[page][:35]
    for page, score in top_pages
]

scores = [score for page, score in top_pages]

plt.figure(figsize=(10, 6))
plt.barh(labels[::-1], scores[::-1])
plt.xlabel("PageRank Score")
plt.ylabel("Web Page")
plt.title("PageRank Ranking of Collected Web Pages")
plt.tight_layout()
plt.show()



import numpy as np

# Adjacency matrix
graph = np.array([
    [0, 1, 1],
    [1, 0, 1],
    [1, 0, 0]
])

n = len(graph)
d = 0.85

# Initialize ranks
rank = np.ones(n) / n

# PageRank iterations
for i in range(100):
    new_rank = np.ones(n) * (1 - d) / n

    for j in range(n):
        if sum(graph[j]) != 0:
            for k in range(n):
                if graph[j][k] == 1:
                    new_rank[k] += d * rank[j] / sum(graph[j])

    rank = new_rank

print("PageRank:")
for i in range(n):
    print("Page", i + 1, ":", round(rank[i], 4))
